        parallel_state.set_virtual_pipeline_model_parallel_rank(model_chunk_id)

        print_all_ranks(f"[FWD {i}] MB {microbatch_id}, VR {model_chunk_id}")

        # Use helper
        output_tensor = forward_step_helper(microbatch_id, None)

        # Send to next stage
        if not parallel_state.is_pipeline_last_stage():
            _profile_p2p_comm(
                p2p_communication.send_forward, 'send_forward', output_tensor, config
            )

        # Collect loss
        if parallel_state.is_pipeline_last_stage() and output_tensor is not None:
            losses_reduced.append(output_tensor)

    print_all_ranks(f"[FWD DONE] Processed {len(microbatch_idx)} forwards")

    # ===== PHASE 2: ALL BACKWARD PASSES =====
    if not forward_only:
        print_all_ranks("=== PHASE 2: All Backwards ===")

        for i, microbatch_id in enumerate(microbatch_idx_b):
            # Handle sync markers
            if microbatch_id == -1:
                print_all_ranks(f"[BWD {i}] SYNC MARKER")
                enable_grad_sync()
                for chunk_id in range(len(model)):
                    if chunk_id not in synchronized_model_chunks:
                        allreduce_gradients(model[chunk_id])
                        synchronized_model_chunks.add(chunk_id)
                disable_grad_sync()
                continue

            model_chunk_id = get_model_chunk_id(microbatch_id, pipeline_parallel_size)
            parallel_state.set_virtual_pipeline_model_parallel_rank(model_chunk_id)

            print_all_ranks(f"[BWD {i}] MB {microbatch_id}, VR {model_chunk_id}")

            # Use helper
            input_tensor_grad = backward_step_helper(microbatch_id)

            # Send to previous stage

                _profile_p2p_comm(
                    p2p_communication.send_backward, 'send_backward', input_tensor_grad, config
                )

        print_all_ranks(f"[BWD DONE] Processed backwards")

    print_all_ranks("=== CHIMERA DONE ===")
